<script>

  function updateAvatar() {

    const name = document.getElementById('studentName').value || 'student';

    const style = document.getElementById('intelligence').value;

    const avatarImg = document.getElementById('avatar');

    

    // Generate avatar URL from DiceBear API

    avatarImg.src = `https://api.dicebear.com/7.x/${style}/svg?seed=${encodeURIComponent(name)}`;

  }


  function downloadAvatar() {

    const avatarImg = document.getElementById('avatar');

    const name = document.getElementById('studentName').value || 'avatar';

    

    fetch(avatarImg.src)

      .then(response => response.blob())

      .then(blob => {

        const url = window.URL.createObjectURL(blob);

        const a = document.createElement('a');

        a.href = url;

        a.download = `${name}-avatar.svg`;

        document.body.appendChild(a);

        a.click();

        a.remove();

      });

  }

</script>
